import {
  REF_COOKIE_MAX_AGE_SEC,
  REF_COOKIE_NAME,
  REF_CODE_LENGTH,
  REF_STORAGE_KEY,
} from "./constants";

const REF_CODE_RE = /^[0-9a-f]{8}$/;

export function normalizeRefCode(raw: string | null | undefined): string | null {
  if (!raw) return null;
  const code = raw.trim().toLowerCase();
  if (!REF_CODE_RE.test(code)) return null;
  return code;
}

export function walletToRefCode(wallet: string): string {
  return wallet.slice(2, 2 + REF_CODE_LENGTH).toLowerCase();
}

export function isSelfReferral(wallet: string, refCode: string): boolean {
  return walletToRefCode(wallet) === refCode.toLowerCase();
}

export function persistRefCode(code: string): void {
  if (typeof window === "undefined") return;
  try {
    localStorage.setItem(REF_STORAGE_KEY, code);
  } catch {
    // ignore
  }
  document.cookie = `${REF_COOKIE_NAME}=${encodeURIComponent(code)}; path=/; max-age=${REF_COOKIE_MAX_AGE_SEC}; SameSite=Lax`;
}

export function readStoredRefCode(): string | null {
  if (typeof window === "undefined") return null;
  try {
    const fromStorage = localStorage.getItem(REF_STORAGE_KEY);
    const normalized = normalizeRefCode(fromStorage);
    if (normalized) return normalized;
  } catch {
    // ignore
  }
  const match = document.cookie.match(
    new RegExp(`(?:^|; )${REF_COOKIE_NAME}=([^;]*)`)
  );
  if (match?.[1]) {
    return normalizeRefCode(decodeURIComponent(match[1]));
  }
  return null;
}

export function captureRefFromUrl(): string | null {
  if (typeof window === "undefined") return null;
  const params = new URLSearchParams(window.location.search);
  const ref = normalizeRefCode(params.get("ref"));
  if (ref) {
    persistRefCode(ref);
    const url = new URL(window.location.href);
    url.searchParams.delete("ref");
    window.history.replaceState({}, "", url.pathname + url.search + url.hash);
  }
  return ref;
}

}

async function expireStaleWithdrawals(wallet: string): Promise<void> {
  const supabase = getReferralSupabase();
  if (!supabase) return;

  const now = new Date().toISOString();
  const { data: stale } = await supabase
    .from("ref_withdrawals")
    .select("id, wallet_address, amount_usd")
    .eq("wallet_address", wallet)
    .eq("status", "pending_claim")
    .lt("expires_at", now);

  if (!stale?.length) return;

  for (const row of stale) {
    await supabase
      .from("ref_withdrawals")
      .update({ status: "expired" })
      .eq("id", row.id);

    const { data: user } = await supabase
      .from("ref_users")
      .select("available_balance")
      .eq("wallet_address", row.wallet_address)
      .single();

    if (user) {
      await supabase
        .from("ref_users")
        .update({
          available_balance: roundUsd(
            Number(user.available_balance) + Number(row.amount_usd)
          ),
        })
        .eq("wallet_address", row.wallet_address);
    }
  }
}

/** Expire stale pending_claim withdrawals every 15 minutes */
async function expireStale() {
  const now = new Date().toISOString();
  const { data: stale } = await supabase
    .from("ref_withdrawals")
    .select("id, wallet_address, amount_usd")
    .eq("status", "pending_claim")
    .lt("expires_at", now);

  for (const row of stale ?? []) {
    await supabase.from("ref_withdrawals").update({ status: "expired" }).eq("id", row.id);
    const { data: user } = await supabase
      .from("ref_users")
      .select("available_balance")
      .eq("wallet_address", row.wallet_address)
      .single();
    if (user) {
      await supabase
        .from("ref_users")
        .update({
          available_balance:
            Math.round((Number(user.available_balance) + Number(row.amount_usd)) * 100) / 100,
        })
        .eq("wallet_address", row.wallet_address);
    }
  }
}
